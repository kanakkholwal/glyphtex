\section{Highlights}
\begin{itemize}
  \item \textbf{Offline mode.} Projects now open and save with no connection, and sync when you are back online.
  \item \textbf{Shared templates.} Teams can publish templates that everyone in the workspace can reuse.
  \item \textbf{Faster search.} Results now appear as you type, even in large workspaces.
\end{itemize}

\release{4.2}{28 September 2026}{Offline mode and shared templates}
\begin{itemize}
  \item \new\ Work offline: edits are stored locally and synced later.
  \item \new\ Shared templates for teams, with a gallery view.
  \item \improved\ Export to PDF keeps page numbers and bookmarks.
  \item \fixed\ Comments no longer disappear after renaming a file.
  \item \fixed\ Dark theme now applies to the settings dialog.
\end{itemize}

\release{4.1.1}{16 September 2026}{Stability fixes}
\begin{itemize}
  \item \fixed\ Crash when pasting very large tables.
  \item \fixed\ Keyboard shortcuts on some non English layouts.
  \item \improved\ Start up time reduced by about 30\%.
\end{itemize}

\release{4.1}{3 September 2026}{Search and performance}
\begin{itemize}
  \item \new\ Instant search with filters for owner and date.
  \item \new\ Pin files to the top of the sidebar.
  \item \improved\ Scrolling in long documents is smoother.
  \item \deprecated\ The legacy API version 1 will be removed in version 5.0.
\end{itemize}
