\chapter{Installation}
\section{Package contents}
\begin{itemize}[nosep]
  \item Filtration unit with pre-fitted cartridge
  \item Wall bracket and four M6 screws
  \item Inlet and outlet hoses, 1.5\,m each
\end{itemize}

\section{Mounting the unit}
\begin{procedure}{Mount the unit on a wall}\label{proc:mount}
  \item Mark the four hole positions using the supplied template.
  \item Drill 8\,mm holes and insert wall plugs.
  \item Fix the bracket with the M6 screws.
  \result{The bracket is level and does not move when pushed.}
  \item Hang the unit on the bracket until it clicks.
\end{procedure}

\begin{notebox}
Leave at least 30\,cm below the unit so the cartridge can be removed later.
\end{notebox}

\chapter{Maintenance}
\section{Replacing the cartridge}
Replace the cartridge every six months or when the indicator turns red. Complete Procedure~\ref{proc:mount} first if the unit is being reinstalled.

\begin{procedure}{Replace the filter cartridge}
  \item Switch off power and close the inlet valve.
  \item Place a tray under the unit and turn the housing a quarter turn anticlockwise.
  \item Remove the old cartridge and fit the new one with the arrow pointing up.
  \item Refit the housing, open the valve slowly and check for leaks.
  \result{The indicator shows green after about ten seconds.}
\end{procedure}

\section{Reading the service log}
Connect to the service port and run:
\begin{lstlisting}
af300 log -n 20
\end{lstlisting}
