\question{4}
Simplify fully: $\dfrac{6x^2 y}{9xy^3}$.
\answer[2]{$\dfrac{2x}{3y^2}$}

\question{6}
Solve the simultaneous equations
\[ 3a + 2b = 16, \qquad a - b = 2. \]
\answer[3]{From the second equation $a = b + 2$. Substituting: $3(b+2) + 2b = 16$, so $5b = 10$, giving $b = 2$ and $a = 4$.}

\question{5}
A jacket costs 80 GBP. Its price is increased by 15\% and then reduced by 10\% in a sale.
\begin{enumerate}[label=(\alph*)]
  \item Find the sale price. \hfill [3]
  \item Find the overall percentage change. \hfill [2]
\end{enumerate}
\answer[3]{(a) $80 \times 1.15 \times 0.9 = 82.80$ GBP. \quad (b) $82.80 / 80 = 1.035$, an increase of $3.5\%$.}

\question{8}
The right-angled triangle $ABC$ has $AB = 7$ cm, $BC = 24$ cm and the right angle at $B$.
\begin{enumerate}[label=(\alph*)]
  \item Calculate the length $AC$. \hfill [3]
  \item Calculate angle $BAC$ to 1 decimal place. \hfill [5]
\end{enumerate}
\answer[4]{(a) $AC = \sqrt{7^2 + 24^2} = \sqrt{625} = 25$ cm. \quad (b) $\tan(BAC) = 24/7$, so $BAC = 73.7^\circ$.}

\question{7}
Expand and simplify $(2x - 3)(x + 5)$, then solve $(2x - 3)(x + 5) = 0$.
\answer[3]{$2x^2 + 7x - 15$. Solutions: $x = 1.5$ or $x = -5$.}

\question{10}
A bag contains 5 red and 3 blue counters. Two counters are taken at random without replacement.
\begin{enumerate}[label=(\alph*)]
  \item Draw a tree diagram to show the outcomes. \hfill [4]
  \item Find the probability that both counters are the same colour. \hfill [6]
\end{enumerate}
\answer[5]{(b) $P(\text{RR}) = \tfrac{5}{8}\cdot\tfrac{4}{7} = \tfrac{20}{56}$ and $P(\text{BB}) = \tfrac{3}{8}\cdot\tfrac{2}{7} = \tfrac{6}{56}$, so $P(\text{same}) = \tfrac{26}{56} = \tfrac{13}{28}$.}
